Remote target over SSH (SRS UC-2).

Uses Windows' own OpenSSH client (ssh.exe) with key authentication only (BatchMode: ssh never asks for a password, FR-20). Nothing about the login is stored except the host, user, port and key-file path the user typed.

connect() copies remote_agent.py + workloads.py + nvgpu.py + pynvml.py into a fresh temp folder on the machine, starts the agent there and waits for its hardware report. After that:

  • RemotePoller stands in for telemetry.Poller (latest / sources / info),
  • RemoteProc stands in for the local workload process (stdout / poll / terminate), so main.py drives a remote test exactly like a local one. The app pings the agent every second; the agent stops its workload and exits after 10 s without contact (FR-22) and deletes its folder when it exits (FR-23).

 44def ssh_runner(host, user, key, port=22):
 45    """argv builder for one command on host over SSH (key login only)."""
 46    base = ["ssh", "-T", "-o", "BatchMode=yes", "-o", "IdentitiesOnly=yes", "-o", "ConnectTimeout=10",
 47            "-o", "ServerAliveInterval=3", "-o", "ServerAliveCountMax=3",
 48            "-o", "StrictHostKeyChecking=accept-new", "-p", str(int(port)), "-i", key, f"{user}@{host}"]
 49    return lambda remote_cmd: base + [remote_cmd]

 52def _bundle():
 53    files = {}
 54    for n in AGENT_FILES:
 55        with open(os.path.join(HERE, n), encoding="utf-8") as f:
 56            files[n] = f.read()
 57    try:  # NVML bindings: one pure-Python file, so the remote machine needs no pip install
 58        import pynvml
 59        with open(pynvml.__file__, encoding="utf-8") as f:
 60            files["pynvml.py"] = f.read()
 61    except (ImportError, OSError):
 62        pass
 63    return json.dumps(files)

 66def _ssh_error(stderr, host, port, key):
 67    text = (stderr or "").strip()
 68    low = text.lower()
 69    if "permission denied" in low:
 70        return f"Login failed: {host} didn't accept the key {key}."
 71    if "could not resolve hostname" in low or "name or service not known" in low:
 72        return f"Can't find {host} (the name doesn't resolve)."
 73    if "timed out" in low:
 74        return f"No answer from {host} on port {port} (connection timed out)."
 75    if "connection refused" in low:
 76        return f"{host} refused the connection on port {port} - is the SSH server running?"
 77    if "host key verification failed" in low or "identification has changed" in low:
 78        return (f"{host}'s host key is not the one seen before. If the machine was rebuilt, remove its "
 79                f"line from ~/.ssh/known_hosts; otherwise this may not be your machine.")
 80    lines = [l for l in text.splitlines() if l.strip()]
 81    return f"SSH failed: {lines[-1] if lines else 'no error message'}"

 84class RemoteProc:
 85    """The workload running on the remote machine, shaped like subprocess.Popen."""
 86    pid = None
 87
 88    def __init__(self, target):
 89        self.target = target
 90        self.q = queue.Queue()
 91        self.returncode = None
 92
 93    @property
 94    def stdout(self):
 95        while True:
 96            line = self.q.get()
 97            if line is None:
 98                return
 99            yield line
100
101    def poll(self):
102        return self.returncode
103
104    def _finish(self, code):
105        if self.returncode is None:
106            self.returncode = code
107            self.q.put(None)
108
109    def terminate(self, timeout=5.0):
110        """Ask the agent to stop the workload; True if it didn't confirm in time."""
111        if self.returncode is not None:
112            return False
113        self.target.send(cmd="stop")
114        end = time.monotonic() + timeout
115        while self.returncode is None and time.monotonic() < end and self.target.alive:
116            time.sleep(0.05)
117        forced = self.returncode is None
118        self._finish(-9)
119        return forced

155class RemoteTarget:
156    def __init__(self, host, user, key, port=22, runner=None):
157        self.host, self.user, self.key, self.port = host, user, key, port
158        self.label = f"{user}@{host}" if user else host
159        self.uses_ssh = runner is None
160        self.run = runner or ssh_runner(host, user, key, port)
161        self.dir = None
162        self.agent = None
163        self.alive = False
164        self.hw = None
165        self.latest, self.info, self.sources = {}, {}, {}
166        self.proc = None
167        self.lost_reason = None
168        self._hello = queue.Queue()
169        self._wlock = threading.Lock()
170
171    # ---------------------------------------------------------------- connect / close
172    def connect(self):
173        """Blocking. Returns the remote hardware report; raises RemoteError with a
174        message for the user."""
175        if self.uses_ssh and not shutil.which("ssh"):
176            raise RemoteError("Windows' OpenSSH client (ssh.exe) isn't installed. Add it under "
177                              "Settings > System > Optional features > OpenSSH Client.")
178        if self.key and not os.path.isfile(self.key):
179            raise RemoteError(f"Key file not found: {self.key}")
180        try:
181            r = subprocess.run(self.run(_UNPACK), input=_bundle(), capture_output=True, text=True,
182                               timeout=45, creationflags=NO_WINDOW)
183        except subprocess.TimeoutExpired:
184            raise RemoteError(f"{self.host} didn't finish the setup within 45 s.")
185        except OSError as e:
186            raise RemoteError(f"Can't run ssh: {e}")
187        if "HCV_NO_PYTHON" in r.stdout:
188            raise RemoteError(f"Python 3 isn't installed on {self.host} (needed: python3 {MIN_PYTHON[0]}."
189                              f"{MIN_PYTHON[1]} or newer).")
190        m = re.search(r"HCV_DIR (\S+) (\d+)\.(\d+)", r.stdout)
191        if r.returncode != 0 or not m:
192            raise RemoteError(_ssh_error(r.stderr, self.host, self.port, self.key) if r.returncode == 255
193                              else f"Setting up the agent failed: {(r.stderr or r.stdout).strip()[-300:]}")
194        if not DIR_RE.match(m.group(1)):
195            raise RemoteError(f"Unexpected agent folder on {self.host}: {m.group(1)}")
196        self.dir = m.group(1)
197        if (int(m.group(2)), int(m.group(3))) < MIN_PYTHON:
198            self._remove_dir()
199            raise RemoteError(f"{self.host} has Python {m.group(2)}.{m.group(3)}; the agent needs "
200                              f"{MIN_PYTHON[0]}.{MIN_PYTHON[1]} or newer.")
201        try:
202            self.agent = subprocess.Popen(
203                self.run(f"cd '{self.dir}' && exec python3 -u remote_agent.py"),
204                stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True,
205                bufsize=1, creationflags=NO_WINDOW)
206        except OSError as e:
207            self._remove_dir()
208            raise RemoteError(f"Can't start the agent: {e}")
209        self.alive = True
210        threading.Thread(target=self._read, daemon=True).start()
211        try:
212            self.hw = self._hello.get(timeout=25)
213        except queue.Empty:
214            self.hw = None
215        if not self.hw:
216            err = self.agent.stderr.read() if self.agent.poll() is not None else ""
217            self.close()
218            raise RemoteError(f"The agent on {self.host} didn't report its hardware. "
219                              f"{err.strip()[-300:]}".strip())
220        threading.Thread(target=self._ping, daemon=True).start()
221        return self.hw
222
223    def close(self):
224        """Stop the agent (it deletes its own folder) and remove the folder again
225        in case it couldn't (FR-23). Doesn't wait for the cleanup."""
226        was_alive, self.alive = self.alive, False
227        if self.proc:
228            self.proc._finish(-9)
229        if self.agent:
230            if was_alive:
231                self.send(cmd="quit", force=True)
232            try:
233                self.agent.stdin.close()
234            except OSError:
235                pass
236            try:
237                self.agent.wait(timeout=3)
238            except subprocess.TimeoutExpired:
239                self.agent.kill()
240        self._remove_dir(wait=False)
241
242    def _remove_dir(self, wait=True):
243        if not self.dir or not DIR_RE.match(self.dir):
244            return
245        try:
246            p = subprocess.Popen(self.run(f"rm -rf -- '{self.dir}'"), stdin=subprocess.DEVNULL,
247                                 stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, creationflags=NO_WINDOW)
248            if wait:
249                p.wait(timeout=20)
250        except (OSError, subprocess.TimeoutExpired):
251            pass
252        self.dir = None
253
254    # ---------------------------------------------------------------- messages
255    def send(self, force=False, **msg):
256        if not (self.alive or force) or not self.agent:
257            return False
258        with self._wlock:
259            try:
260                self.agent.stdin.write(json.dumps(msg) + "\n")
261                self.agent.stdin.flush()
262                return True
263            except (OSError, ValueError):
264                return False
265
266    def _ping(self):
267        while self.alive:
268            self.send(cmd="ping")
269            time.sleep(1.0)
270
271    def _read(self):
272        for line in self.agent.stdout:
273            try:
274                m = json.loads(line)
275            except ValueError:
276                continue
277            if "tel" in m:
278                self.latest = dict(m["tel"], t=time.monotonic())
279                self.info = m.get("info") or {}
280            elif "wl" in m:
281                if self.proc:
282                    self.proc.q.put(m["wl"])
283            elif "ended" in m:
284                if self.proc:
285                    self.proc._finish(m["ended"])
286            elif "hello" in m:
287                self._hello.put(m["hello"])
288            elif "sources" in m:
289                self.sources = m["sources"]
290        # the SSH session ended
291        if self.alive:
292            self.lost_reason = f"Connection to {self.host} lost."
293        self.alive = False
294        self._hello.put(None)
295        if self.proc and self.proc.returncode is None:
296            self.proc.q.put(json.dumps({"error": f"connection to {self.host} lost"}))
297            self.proc._finish(-1)
298
299    # ---------------------------------------------------------------- used by main.py
300    def poller(self, gpu_index, disk):
301        return RemotePoller(self, gpu_index, disk)
302
303    def start(self, script, seconds, disk=None, command=None):
304        self.proc = RemoteProc(self)
305        if not self.send(cmd="start", script=script, seconds=seconds, disk=disk, command=command):
306            raise OSError(f"connection to {self.host} lost")
307        return self.proc
