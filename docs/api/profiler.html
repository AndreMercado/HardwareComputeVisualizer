profiler

Line profiler for the user's own Python code (drives the VS Code extension's boxes).

python profiler.py [--out PROFILE.json] [--memory] [--interval MS] [--root PROJECT]
                   [--cwd FOLDER] script.py [script args...]

"Your code" is every file in the project folder (--root; default: the nearest folder above the script with .git, pyproject.toml, requirements.txt, ...), minus virtual environments, site-packages and the Python install. A file of a package that uses relative imports (or a package's __main__.py) runs like python -m pkg.module.

Runs script.py as __main__ and, every few ms, notes which line of your code is running: the innermost frame in a file of your project, so time spent inside numpy, PyTorch or the standard library is charged to the line of yours that called it. Each sample is weighted by the time since the previous one (wall-clock time). At the same moments it records what the hardware was doing: how many cores this program kept busy (100 = one core; other programs don't count), the whole CPU, and the GPU - so a hot spot can say "one core busy, GPU idle while here".

The result is written to PROFILE.json (default: .hcv/profile.json next to the script) every second while the script runs - the editor updates live - and once more at the end. --memory also records, for each line, the most memory its allocations held at once (tracemalloc, checked every second; makes the script slower, so times are less exact). Only the main thread is profiled; child processes are not.

Accuracy: a sampler inside the program can only look when Python lets other threads run, so time inside long C calls (sort, regex, json...) lands on the next line, and short functions on their def line. When py-spy is installed (pip install py-spy) it is used for the final numbers instead: it samples from outside the program, sees every line exactly, plus all threads and child processes. The in-process sampler still draws the live boxes while the program runs and records the hardware readings.

 87def _python_dirs():
 88    """Folders of the Python installation (standard library, installed packages): never the user's."""
 89    import sysconfig
 90    dirs = {sys.prefix, sys.base_prefix, sys.exec_prefix, getattr(sys, "base_exec_prefix", sys.prefix)}
 91    for key in ("stdlib", "platstdlib", "purelib", "platlib"):
 92        try:
 93            dirs.add(sysconfig.get_paths()[key])

 99def project_root(script):
100    """The project a script belongs to: the nearest folder above it with a project marker
101    (.git, pyproject.toml, requirements.txt, .vscode...). Never the home folder or a drive's
102    top; without a marker, the script's own folder."""
103    start = os.path.dirname(os.path.realpath(script))
104    home = _norm(os.path.expanduser("~"))
105    d = start
106    while True:
107        parent = os.path.dirname(d)
108        if parent == d or _norm(d) == home:
109            break
110        if any(os.path.exists(os.path.join(d, m)) for m in ROOT_MARKERS):
111            return d
112        d = parent
113    return start

116def module_name(script):
117    """'pkg.sub.mod' when the script is part of a package and must be run as `python -m` (it
118    uses relative imports, or is a package's __main__.py); else None. Also returns the folder
119    that holds the top package (what `python -m` puts first on sys.path)."""
120    folder, base = os.path.split(os.path.realpath(script))
121    if not os.path.isfile(os.path.join(folder, "__init__.py")):
122        return None, None
123    if base != "__main__.py":
124        try:
125            with open(script, encoding="utf-8", errors="replace") as f:
126                if not re.search(r"^\s*from\s+\.", f.read(), re.M):
127                    return None, None
128        except OSError:
129            return None, None
130    parts = [] if base == "__main__.py" else [os.path.splitext(base)[0]]
131    while os.path.isfile(os.path.join(folder, "__init__.py")):
132        folder, pkg = os.path.split(folder)
133        parts.insert(0, pkg)
134    if not all(p.isidentifier() for p in parts):
135        return None, None
136    return ".".join(parts), folder

140    def __init__(self, script, out, interval_ms=5, memory=False, root=None):
141        self.script = os.path.realpath(script)
142        self.root = _norm(root or project_root(self.script))
143        self.python_dirs = _python_dirs()
144        self.venvs = {}        # folder -> is it a virtual environment (has pyvenv.cfg)?
145        self._write_lock = threading.Lock()
146        self.out = out
147        self.interval = interval_ms / 1000.0
148        self.memory = memory
149        self.lines = {}        # (file, line) -> [seconds, func, {hw sums}, hw samples]
150        self.hw = {}           # latest hardware reading
151        self.hw_peak_gpu = 0.0
152        self.t0 = time.perf_counter()
153        self.sampled = 0.0
154        self.done = threading.Event()
155        self.result = None     # how the script ended
156        self.exit_code = None
157        self.crash = None      # {file, line, error}: where the script failed, in the user's code
158        self.finalizing = False  # True when py-spy will write the final numbers after this process
159        self._mine = {}        # filename -> is it one of the user's files?

161        self.cores = os.cpu_count() or 1

175    def _is_mine(self, filename):
176        hit = self._mine.get(filename)
177        if hit is None:
178            p = _norm(filename) if filename and not filename.startswith("<") else ""
179            hit = bool(p) and (p == self.root or p.startswith(self.root.rstrip(os.sep) + os.sep)) \
180                and not any(s in p for s in SKIP_PARTS) \
181                and not any(p.startswith(d) for d in self.python_dirs) \
182                and p not in {_norm(os.path.join(HERE, f)) for f in OWN_FILES} \
183                and not self._in_venv(p)
184            self._mine[filename] = hit
185        return hit

187    def _in_venv(self, p):
188        """Inside a virtual environment or a tool folder within the project (.venv, venv, .tox...)?"""
189        rel = os.path.relpath(os.path.dirname(p), self.root)
190        if rel == ".":
191            return False
192        d = self.root
193        for part in rel.split(os.sep):
194            if part in SKIP_DIRS:
195                return True
196            d = os.path.join(d, part)
197            hit = self.venvs.get(d)
198            if hit is None:
199                hit = self.venvs[d] = os.path.isfile(os.path.join(d, "pyvenv.cfg")) or \
200                    os.path.isfile(os.path.join(d, "conda-meta", "history"))
201            if hit:
202                return True
203        return False

206    def _sample_loop(self):
207        self.own_threads.add(threading.get_ident())
208        last = time.perf_counter()
209        while not self.done.is_set():
210            time.sleep(self.interval)
211            now = time.perf_counter()
212            dt, last = now - last, now
213            frames = sys._current_frames()
214            hw = self.hw
215            if hw:  # worker threads: what the hardware did while their lines ran (py-spy times them)
216                for ident, fr in frames.items():
217                    if ident == self.main_ident or ident in self.own_threads:
218                        continue
219                    while fr is not None and not self._is_mine(fr.f_code.co_filename):
220                        fr = fr.f_back
221                    if fr is None:
222                        continue
223                    rec = self.thread_hw.setdefault((fr.f_code.co_filename, fr.f_lineno),
224                                                    [dict.fromkeys(LINE_HW, 0.0), dict.fromkeys(LINE_HW, 0.0)])
225                    for k, src in LINE_HW.items():
226                        v = hw.get(src)
227                        if v is not None:
228                            rec[0][k] += v * dt
229                            rec[1][k] += dt
230            frame = frames.get(self.main_ident)
231            while frame is not None and not self._is_mine(frame.f_code.co_filename):
232                frame = frame.f_back
233            self.sampled += dt
234            if frame is None:
235                continue  # interpreter start-up / shutdown, outside the user's code
236            key = (frame.f_code.co_filename, frame.f_lineno)
237            rec = self.lines.get(key)
238            if rec is None:
239                rec = self.lines[key] = [0.0, frame.f_code.co_name, dict.fromkeys(LINE_HW, 0.0),
240                                         dict.fromkeys(LINE_HW, 0.0)]
241            rec[0] += dt
242            hw = self.hw
243            if hw:
244                for k, src in LINE_HW.items():
245                    v = hw.get(src)
246                    if v is not None:
247                        rec[2][k] += v * dt
248                        rec[3][k] += dt
249            self._to_timeline(now - self.t0, key, dt)

251    def _to_timeline(self, t, key, dt):
252        """Adds dt of `key` to the current bucket; a full bucket goes to the timeline as its top line."""
253        start, counts = self._bucket
254        if t - start >= BUCKET_S:
255            if counts:
256                top = max(counts, key=counts.get)
257                idx = self._tl_index.get(top)
258                if idx is None:
259                    idx = self._tl_index[top] = len(self.tl_keys)
260                    self.tl_keys.append(top)
261                if len(self.timeline) < MAX_SAMPLES:
262                    self.timeline.append([round(start, 2), idx])
263            start, counts = start + BUCKET_S * int((t - start) / BUCKET_S), {}
264            self._bucket = (start, counts)
265        if key is not None:
266            counts[key] = counts.get(key, 0.0) + dt

268    def _hardware_loop(self):
269        """Every HW_EVERY_S: what the machine was doing (the readings the visualizer's bottleneck
270        finder uses), as self.hw for the line running now, and in self.samples."""
271        self.own_threads.add(threading.get_ident())
272        try:
273            import psutil
274        except ImportError:
275            self.hw_missing = f"psutil isn't installed in this Python ({sys.executable}): pip install psutil"
276            self.hw_ready.set()
277            return
278        gpu = None
279        try:
280            nvgpu = _private_module("nvgpu")
281            if nvgpu.OK and nvgpu.names():
282                gpu = nvgpu.Gpu(0)
283                gpu.start()
284        except Exception:
285            gpu = None
286        me = psutil.Process()
287        kids = {}
288
289        def own_cpu():
290            """CPU used by this program and its child processes, 100 = one core."""
291            total = me.cpu_percent(None)
292            try:
293                for c in me.children(recursive=True):
294                    p = kids.setdefault(c.pid, c)
295                    total += p.cpu_percent(None)
296            except psutil.Error:
297                pass
298            return total
299
300        def disk_counters():
301            try:
302                return psutil.disk_io_counters()
303            except Exception:
304                return None
305
306        psutil.cpu_percent(None, percpu=True)
307        own_cpu()
308        last_disk, last_t = disk_counters(), time.perf_counter()
309        self.hw_ready.set()
310        while not self.done.is_set():
311            time.sleep(HW_EVERY_S)
312            now = time.perf_counter()
313            cores = psutil.cpu_percent(None, percpu=True)
314            row = {"t": round(now - self.t0, 3), "proc": round(own_cpu(), 1),
315                   "cpu_util": round(sum(cores) / len(cores), 1) if cores else None,
316                   "cpu_core_max": max(cores) if cores else None}
317            try:
318                vm = psutil.virtual_memory()
319                row["ram_util"], row["ram_used"] = vm.percent, round(vm.used / 2**30, 3)
320            except Exception:
321                pass
322            disk = disk_counters()
323            dt = now - last_t
324            if disk is not None and last_disk is not None and dt > 0:
325                row["disk_read"] = round((disk.read_bytes - last_disk.read_bytes) / dt / 1e6, 2)
326                row["disk_write"] = round((disk.write_bytes - last_disk.write_bytes) / dt / 1e6, 2)
327                if getattr(disk, "busy_time", None) is not None:  # Linux
328                    busy_ms = disk.busy_time - last_disk.busy_time
329                else:  # Windows: time spent reading + writing
330                    busy_ms = (disk.read_time + disk.write_time) - (last_disk.read_time + last_disk.write_time)
331                row["disk_util"] = round(max(0.0, min(100.0, busy_ms / 10.0 / dt)), 1)
332            last_disk, last_t = disk, now
333            if gpu is not None and len(self.samples) % 10 == 0 and not self.gpu_mine:
334                self.gpu_mine = self._uses_gpu(gpu, nvgpu.pynvml, me, kids)
335            if gpu is not None:
336                try:
337                    for k, v in gpu.sample().items():
338                        if v is not None:
339                            row[k] = round(v, 3) if isinstance(v, float) else v
340                    if row.get("gpu_util") is not None:
341                        self.hw_peak_gpu = max(self.hw_peak_gpu, row["gpu_util"])
342                except Exception:
343                    pass
344            self.hw = row
345            if len(self.samples) < MAX_SAMPLES:
346                self.samples.append(row)
347        if gpu is not None:
348            gpu.stop()

350    def _uses_gpu(self, gpu, pynvml, me, kids):
351        """Is this program among the GPU's processes? (Other programs - the desktop, a browser - keep
352        the GPU partly busy; that must not read as this program's GPU work.)"""
353        pids = set()
354        for fn in ("nvmlDeviceGetComputeRunningProcesses", "nvmlDeviceGetGraphicsRunningProcesses"):
355            try:
356                pids |= {p.pid for p in getattr(pynvml, fn)(gpu.handle)}
357            except Exception:
358                pass
359        if not pids:
360            return self.gpu_mine  # not supported here: undecided
361        return bool(pids & ({me.pid} | set(kids)))
362
363    def gpu_used(self):
364        """This program used the GPU: NVML lists it, else (NVML can't list processes) its memory or
365        load rose clearly above where it started."""
366        if self.gpu_mine is not None:
367            return self.gpu_mine
368        g = [r["gpu_util"] for r in self.samples if num(r.get("gpu_util"))]
369        v = [r["gpu_vram"] for r in self.samples if num(r.get("gpu_vram"))]
370        base = sorted(g[:10])[len(g[:10]) // 2] if g else 0
371        return bool((v and max(v) - v[0] >= 0.15) or (g and max(g) - base >= 40))

385        # the writer thread and the end of the run both write: one at a time, and never an old

387        with self._write_lock:

392    def _write(self, running):
393        final = not running
394        if not running and self.finalizing:
395            running = True  # the parent (py-spy) writes the final, exact numbers next
396        elapsed = time.perf_counter() - self.t0
397        total = sum(r[0] for r in list(self.lines.values())) or 1e-9
398        lines = []
399        for (fname, line), (secs, func, hw, hw_t) in sorted(list(self.lines.items()), key=lambda kv: -kv[1][0]):
400            item = {"file": os.path.realpath(fname), "line": line, "func": func, "seconds": round(secs, 4),
401                    "share": round(secs / total, 5)}
402            avg = {k: round(hw[k] / hw_t[k], 1) for k in hw if hw_t[k] > 0}
403            if avg:
404                item["hw"] = avg
405            lines.append(item)
406        data = {"version": 1, "script": self.script, "running": running, "elapsed_s": round(elapsed, 2),
407                "profiled_s": round(total, 3), "interval_ms": self.interval * 1000, "cores": self.cores,
408                "gpu_used": self.gpu_used(), "result": self.result, "exit_code": self.exit_code,
409                "engine": "builtin", "crash": self.crash, "lines": lines[:500], "updated": time.time(),
410                "started_epoch": self.started_epoch, "root": self.root, "hw_missing": self.hw_missing}
411        if final:
412            # the whole recording, and what it means: bottlenecks found in the hardware samples,
413            # tied to the lines that were running while each one held
414            self._to_timeline(elapsed + BUCKET_S, None, 0.0)
415            data["timeline"] = self.timeline
416            data["timeline_keys"] = [[os.path.realpath(f), ln] for f, ln in self.tl_keys]
417            data["samples"] = self.samples
418            data["thread_hw"] = [{"file": os.path.realpath(f), "line": ln,
419                                  "hw": {k: round(sums[k] / secs[k], 1) for k in sums if secs[k] > 0}}
420                                 for (f, ln), (sums, secs) in self.thread_hw.items() if any(secs.values())]
421            try:
422                link_findings(data, self.cores)
423            except Exception as e:  # never lose the profile over the extras
424                data["findings_error"] = f"{type(e).__name__}: {e}"
425        if self.memory:
426            if not running:
427                self._memory_check()
428            top = sorted(self.mem_peak.items(), key=lambda kv: -kv[1])[:50]
429            data["memory"] = [{"file": os.path.realpath(f), "line": ln, "bytes": b} for (f, ln), b in top]
430        make_out_dir(self.out)
431        tmp = self.out + ".tmp"
432        with open(tmp, "w", encoding="utf-8") as f:
433            json.dump(data, f)
434        for _ in range(20):  # the editor may be reading the old file: retry briefly
435            try:
436                os.replace(tmp, self.out)
437                break
438            except PermissionError:
439                time.sleep(0.05)

441    def _memory_check(self):
442        import tracemalloc
443        if not tracemalloc.is_tracing():
444            return
445        for stat in tracemalloc.take_snapshot().statistics("lineno"):
446            fr = stat.traceback[0]

513def _states_by_line(data, samples, cores, t_offset=0.0):
514    """{(file, line): {pattern: seconds}} - which bottleneck patterns held while each line ran.
515    `samples` t are seconds since their own start; t_offset moves them onto the profile's clock."""

662def pyspy_lines(raw_path, script, root=None):
663    """py-spy's collapsed stacks -> {(file, line): [seconds, func]} for the user's innermost frame
664    of every sample (all threads and processes; waiting time included)."""
665    judge = Profiler(script, "", 5, root=root)  # only for its _is_mine()
666    lines, threads = {}, set()
667    with open(raw_path, encoding="utf-8", errors="replace") as f:
668        for row in f:
669            stack, _, count = row.rstrip("\n").rpartition(" ")
670            if not count.isdigit():
671                continue
672            frames = stack.split(";")
673            threads.add(";".join(frames[:2]))
674            for fr in reversed(frames):
675                m = _FRAME.match(fr)
676                if m and judge._is_mine(m.group(2)):
677                    key = (os.path.realpath(m.group(2)), int(m.group(3)))
678                    rec = lines.setdefault(key, [0.0, m.group(1).split(".")[-1]])
679                    rec[0] += int(count) / PYSPY_RATE
680                    break
681    return lines, len(threads)

684def run_with_pyspy(pyspy, a, out):
685    """py-spy runs this file again as the program (--child: live boxes + hardware), then the
686    exact line times from py-spy are merged with the child's hardware readings."""
687    raw = out + ".pyspy.txt"
688    child = [sys.executable, os.path.abspath(__file__), "--child", "--out", out, "--interval", str(a.interval),
689             "--root", a.root]
690    child += (["--memory"] if a.memory else []) + [a.script] + a.args
691    cmd = [pyspy, "record", "--rate", str(PYSPY_RATE), "--idle", "--threads", "--subprocesses",
692           "--full-filenames", "--format", "raw", "-o", raw, "--"] + child
693    make_out_dir(out)
694    try:
695        rc = subprocess.call(cmd)
696    except KeyboardInterrupt:  # Ctrl+C reaches the program too: it saves "stopped"; finish up below
697        rc = 130
698    live = None
699    try:
700        with open(out, encoding="utf-8") as f:
701            live = json.load(f)
702        exact, n_threads = pyspy_lines(raw, a.script, a.root)
703    except (OSError, ValueError) as e:
704        print(f"[hcv] py-spy result unusable ({e}); keeping the built-in numbers", file=sys.stderr)
705        exact, n_threads = {}, 1
706        if live is None:
707            return rc
708    finally:
709        try:
710            os.remove(raw)
711        except OSError:
712            pass
713    if not exact:
714        live["running"] = False
715        exact_lines = live["lines"]
716    else:
717        # hardware per line comes from the in-process sampler: same line, else same function
718        hw_line = {(os.path.normcase(l["file"]), l["line"]): l.get("hw")
719                   for l in (live.get("thread_hw") or []) + live["lines"] if l.get("hw")}
720        st_line = {(os.path.normcase(l["file"]), l["line"]): l.get("states") for l in live["lines"] if l.get("states")}
721        hw_func = {}
722        for l in live["lines"]:
723            if l.get("hw"):
724                hw_func.setdefault((os.path.normcase(l["file"]), l["func"]), l["hw"])
725        total = sum(v[0] for v in exact.values()) or 1e-9
726        exact_lines = []
727        for (fname, line), (secs, func) in sorted(exact.items(), key=lambda kv: -kv[1][0]):
728            item = {"file": fname, "line": line, "func": func, "seconds": round(secs, 4),
729                    "share": round(secs / total, 5)}
730            hw = hw_line.get((os.path.normcase(fname), line)) or hw_func.get((os.path.normcase(fname), func))
731            if hw:
732                item["hw"] = hw
733            st = st_line.get((os.path.normcase(fname), line))
734            if st:
735                item["states"] = st
736            exact_lines.append(item)
737        live.update(profiled_s=round(total, 3), threads=n_threads)
738    live.update(lines=exact_lines[:500], running=False, engine="py-spy" if exact else "builtin",
739                updated=time.time())
740    tmp = out + ".tmp"
741    with open(tmp, "w", encoding="utf-8") as f:
742        json.dump(live, f)
743    os.replace(tmp, out)
744    code = live.get("exit_code")
745    return code if isinstance(code, int) else rc
